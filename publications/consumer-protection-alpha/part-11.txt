function unit(node,root){

function contentCell(container){
 if(container.matches('tr.subsection,tr.detail')){
  return [...container.children].filter(e=>e.tagName==='TD').at(-1)||null;
 }

  const row=container.querySelector(':scope > tbody > tr, :scope > tr');
  return row?[...row.children].filter(e=>e.tagName==='TD').at(-1)||null:null;

}
function introductions(target,root){
 const entries=[];let node=target.parentElement;
 while(node&&node!==root){
  if(node.matches?.('tr.subsection[id],tr.detail[id],table.article[id]')){
   const content=contentCell(node);
   if(content?.contains(target)){
    const first=[...content.childNodes].find(n=>n.nodeType===1&&n.matches?.('table,ul,ol'));
    const branch=[...content.childNodes].find(n=>n.nodeType===1&&n.contains?.(target));
    if(first&&branch&&branch.matches?.('table,ul,ol')){
     const range=document.createRange();range.setStart(content,0);range.setEndBefore(first);
     if(norm(range.toString()))entries.unshift({owner:node,content,boundary:first,range});
    }
   }
  }

 }
 return entries;
}
